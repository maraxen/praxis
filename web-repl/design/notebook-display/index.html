Standard curve on the STARlet

Moves 50, 100 and 150 µL from the source plate into columns 1 to 3 of the assay plate, with fresh tips for each column. Runs on the simulator. To use the machine, swap the backend for STARBackend().

1 
from pylabrobot.liquid_handling import LiquidHandler
from pylabrobot.liquid_handling.backends import LiquidHandlerChatterboxBackend
from pylabrobot.resources.hamilton import STARLetDeck, TIP_CAR_480_A00, PLT_CAR_L5AC_A00
from pylabrobot.resources.hamilton import hamilton_96_tiprack_300uL_filter
from pylabrobot.resources.corning.plates import Cor_96_wellplate_360ul_Fb

deck = STARLetDeck()
lh = LiquidHandler(backend=LiquidHandlerChatterboxBackend(num_channels=8), deck=deck)
await lh.setup()

tip_car = TIP_CAR_480_A00(name="tip_carrier")
tip_car[0] = tips = hamilton_96_tiprack_300uL_filter(name="tips_300")
deck.assign_child_resource(tip_car, rails=3)

plate_car = PLT_CAR_L5AC_A00(name="plate_carrier")
plate_car[0] = source = Cor_96_wellplate_360ul_Fb(name="source")
plate_car[1] = assay = Cor_96_wellplate_360ul_Fb(name="assay")
deck.assign_child_resource(plate_car, rails=9)
deck

from praxis.display import ledger

with ledger(lh) as run:
    for col, vol in [(1, 50), (2, 100), (3, 150)]:
        await lh.pick_up_tips(tips[f"A{col}:H{col}"])
        await lh.aspirate(source[f"A{col}:H{col}"], vols=[vol] * 8)
        await lh.dispense(assay[f"A{col}:H{col}"], vols=[vol] * 8)
        await lh.discard_tips()
run
